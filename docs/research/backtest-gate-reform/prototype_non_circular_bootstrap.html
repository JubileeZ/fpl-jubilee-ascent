# Non-Wrapping Moving Block Bootstrap Implementation (ADR 0064 / Ticket #171)
def block_bootstrap_win_probability(deltas, *, draws=20_000, block=5, seed=0):
    values = np.asarray(deltas, dtype=float)
    n = len(values)
    size = min(block, n)
    k = -(-n // size)  # ceil(n / size) blocks
    num_starts = n - size + 1
    starts = np.random.default_rng(seed).integers(0, num_starts, size=(draws, k))
    idx = (starts[:, :, None] + np.arange(size)).reshape(draws, -1)
    return float((values[idx].mean(axis=1) > 0).mean())
